cmake_minimum_required(VERSION 3.24)
project(native_media_core VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(native_media_core SHARED ffi/media.cpp)
target_include_directories(native_media_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/ffi)
target_compile_definitions(native_media_core PRIVATE MEDIA_BUILD)
set_target_properties(native_media_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(native_media_core PRIVATE /W4 /WX)
else()
  target_compile_options(native_media_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(media_abi_test tests/abi_test.c)
  target_link_libraries(media_abi_test PRIVATE native_media_core)
  add_test(NAME media_abi COMMAND media_abi_test)
endif()
